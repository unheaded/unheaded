	defer m.Stop()

	var counter int
	var mu sync.Mutex

	m.RegisterMutation("increment", func(ctx context.Context, input any) Result[any] {
		mu.Lock()
		counter++
		mu.Unlock()
		return Ok[any](counter)
	})

	var wg sync.WaitGroup
	ctx := context.Background()

	for i := 0; i < 100; i++ {
		wg.Add(1)
		go func() {
			defer wg.Done()
			_ = m.Mutate(ctx, "increment", nil)
		}()
	}

	wg.Wait()

	if counter != 100 {
		t.Errorf("counter = %d, want 100", counter)
	}
}

func TestMonad_ConcurrentTransactions_Isolated(t *testing.T) {
	m := createTestMonad(t)
	defer m.Stop()

	var wg sync.WaitGroup
	ctx := context.Background()
	results := make(chan error, 10)

	for i := 0; i < 10; i++ {
		wg.Add(1)
		go func(id int) {
			defer wg.Done()

			tx := m.BeginTransaction(ctx)
			if tx == nil {
				results <- errors.New("nil transaction")
				return
			}

			tx.Mutate("op", id)

			if id%2 == 0 {
				results <- tx.Commit()
			} else {
				results <- tx.Rollback()
			}
		}(i)
	}

	wg.Wait()
	close(results)

	for err := range results {
		if err != nil {
			t.Errorf("transaction error: %v", err)
		}
	}
}

func TestSophia_Decide_WithFactors_WeighsOptions(t *testing.T) {
	s := createTestSophia(t)
	defer s.Stop()

	// Teach Sophia some knowledge to inform decision
	ctx := context.Background()
	_ = s.Learn(ctx, createTestKnowledge("Go", "has", "fast-compilation"))
	_ = s.Learn(ctx, createTestKnowledge("Go", "has", "good-concurrency"))
	_ = s.Learn(ctx, createTestKnowledge("Rust", "has", "memory-safety"))
	_ = s.Learn(ctx, createTestKnowledge("Rust", "has", "steep-learning-curve"))

	decision, err := s.Decide(ctx,
		"Which language for a new service?",
		[]string{"Go", "Rust"},
		[]Factor{
			{Name: "fast-compilation", Weight: 0.8},
			{Name: "good-concurrency", Weight: 0.9},
		},
	)

	if err != nil {
		t.Errorf("Decide failed: %v", err)
	}

	// With these factors, Go should score higher
	if decision.Choice != "Go" {
