DEPLOYMENT WORKFLOW

   bash setup-vllm.sh      # Launch vLLM DeepSeek-R1

5. MONITOR:
   make logs SVC=prometheus     # View logs
   make exec SVC=athena         # Enter container
   make snapshot                # Backup containers

6. CLEANUP:
   make stop               # Stop all containers
   make destroy            # Delete all containers
   make clean              # Delete + profiles + network
