```go
package main

import (
    "encoding/json"
    "os"
    "fmt"
)

type User struct {
    ID   int
    Name string
}

func saveUser(u User) error {
    data, err := json.Marshal(u)
    if err != nil {
        return err
    }
    return os.WriteFile("/tmp/user.json", data, 0644)
}

func main() {
    user := User{ID: 1, Name: "John Doe"}
    err := saveUser(user)
    if err != nil {
        fmt.Println("Error saving user:", err)
        return
    }
    fmt.Println("User saved successfully")
}
```

In this example, the `User` struct is defined with `ID` and `Name` fields, and the `saveUser` function is called within the `main` function to save a user. The error handling is also demonstrated to ensure that any issues are properly reported.
